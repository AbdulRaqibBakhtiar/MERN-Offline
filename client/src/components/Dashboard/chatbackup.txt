// ==========================================
// LIVE CHAT DRAWER - PART 1 OF 9
// ==========================================
import React, { useState, useEffect, useRef } from 'react';
import { MessageSquare, MessageCircle, Send, Paperclip, Smile, Edit2, Trash2, X, Check, Mic, Square, Phone, Video, PhoneOff } from 'lucide-react';
import EmojiPicker from 'emoji-picker-react';
import axios from "axios";

const REACTION_OPTIONS = ['👍', '❤️', '😂', '😮', '😢', '🙏'];

// Standard public STUN servers for WebRTC peer pathway detection
const RTC_CONFIG = { iceServers: [{ urls: 'stun:://google.com' }] };

export const LiveChatDrawer = ({ socket, user }) => {
    const [isOpen, setIsOpen] = useState(false);
    const [message, setMessage] = useState('');
    const [chatLog, setChatLog] = useState([]);
    const [typingUser, setTypingUser] = useState('');
    const [showEmojiPicker, setShowEmojiPicker] = useState(false);
    const [selectedFile, setSelectedFile] = useState(null);
    const [hoveredMessageId, setHoveredMessageId] = useState(null);
    const [editingMessageId, setEditingMessageId] = useState(null);
    const [editInput, setEditInput] = useState('');
    const [toast, setToast] = useState({ show: false, message: '', type: 'info' });
    const [activeUsersList, setActiveUsersList] = useState([]);
    const [reactionsMap, setReactionsMap] = useState({});

    // Audio note recorder states
    const [isRecording, setIsRecording] = useState(false);
    const [recordingDuration, setRecordingDuration] = useState(0);

    // 📹 NEW WEBRTC VIDEO/VOICE CALLING STATES
    const [callState, setCallState] = useState('idle'); // 'idle' | 'ringing' | 'incoming' | 'connected'
    const [callType, setCallType] = useState('video'); // 'voice' | 'video'
    const [remotePeerDetails, setRemotePeerDetails] = useState({ socketId: '', name: '' });
    // ==========================================
    // LIVE CHAT DRAWER - PART 2 OF 9
    // ==========================================
    const messageEndRef = useRef(null);
    const typingTimeoutRef = useRef(null);
    const mediaRecorderRef = useRef(null);
    const audioChunksRef = useRef([]);
    const durationIntervalRef = useRef(null);

    // 📹 WebRTC Stream References
    const localVideoRef = useRef(null);
    const remoteVideoRef = useRef(null);
    const localStreamRef = useRef(null);
    const peerConnectionRef = useRef(null);

    const currentUsername = user?.username || user?.name || "Anonymous";
    const currentUserId = user?._id || user?.id || "guest_user";

    const triggerToast = (msg, type = 'info') => {
        setToast({ show: true, message: msg, type });
        if (type !== 'error') {
            setTimeout(() => setToast({ show: false, message: '', type: 'info' }), 3000);
        }
    };

    useEffect(() => {
        if (!socket || !user) return;
        const join = { room: "global_workspace", username: currentUsername, userId: currentUserId };
        if (socket.connected) socket.emit("join_room", join);
        socket.on("connect", () => socket.emit("join_room", join));
        return () => socket.off("connect");
    }, [socket, user, currentUsername, currentUserId]);
    // ==========================================
    // LIVE CHAT DRAWER - PART 3 OF 9
    // ==========================================
    useEffect(() => {
        if (!isOpen) return;
        const load = async () => {
            try {
                const res = await axios.get("http://localhost:5000/chat/history");
                const raw = Array.isArray(res.data) ? res.data : [];
                const fmt = raw.map(m => ({ ...m, readBy: Array.isArray(m.readBy) ? m.readBy : [] }));
                setChatLog(fmt);
                if (socket && currentUserId !== "guest_user") {
                    fmt.forEach(m => {
                        if (m.senderName !== currentUsername && !m.readBy.some(r => r?.userId === currentUserId)) {
                            socket.emit("mark_as_read", { messageId: m._id, userId: currentUserId, room: "global_workspace" });
                        }
                    });
                }
            } catch (err) { console.error(err); }
        };
        load();
    }, [isOpen, socket, currentUserId, currentUsername]);

    useEffect(() => {
        if (isOpen) setTimeout(() => messageEndRef.current?.scrollIntoView({ behavior: 'smooth' }), 100);
    }, [chatLog, isOpen, typingUser]);
    // ==========================================
    // LIVE CHAT DRAWER - PART 4 OF 4 (MESSAGES)
    // ==========================================
    useEffect(() => {
        if (!socket) return;
        socket.on("receive_message", (m) => {
            setChatLog(prev => [...prev, { ...m, readBy: Array.isArray(m.readBy) ? m.readBy : [] }]);
            if (isOpen && m.senderName !== currentUsername) {
                socket.emit("mark_as_read", { messageId: m._id, userId: currentUserId, room: "global_workspace" });
            }
        });
        socket.on("user_typing", ({ username, isTyping }) => setTypingUser(isTyping ? username : ''));
        socket.on("message_read_update", ({ messageId, readBy }) => {
            setChatLog(prev => prev.map(m => m._id === messageId ? { ...m, readBy } : m));
        });
        socket.on("message_edited", ({ messageId, newMessage }) => {
            setChatLog(prev => prev.map(m => m._id === messageId ? { ...m, message: newMessage } : m));
            triggerToast("Message updated successfully", "success");
        });
        socket.on("message_deleted", ({ messageId }) => {
            setChatLog(prev => prev.filter(m => m._id !== messageId));
            triggerToast("Message deleted", "success");
        });
        socket.on("room_users_update", (users) => setActiveUsersList(users || []));

        socket.on("message_reaction_update", ({ messageId, emoji, username }) => {
            setReactionsMap(prev => {
                const curr = prev[messageId] || [];
                const idx = curr.findIndex(r => r.username === username);
                let updated = [...curr];
                if (idx > -1) {
                    if (updated[idx].emoji === emoji) updated = updated.filter(r => r.username !== username);
                    else updated[idx].emoji = emoji;
                } else { updated.push({ emoji, username }); }
                return { ...prev, [messageId]: updated };
            });
        });
        // ==========================================
        // LIVE CHAT DRAWER - PART 5 OF 9 (rtc handlers)
        // ==========================================
        // Handle WebRTC connection alerts dropping from backend socket routers
        socket.on("incoming_call_signal", ({ fromSocketId, callerName, type }) => {
            setCallState('incoming');
            setCallType(type);
            setRemotePeerDetails({ socketId: fromSocketId, name: callerName });
        });

        socket.on("call_accepted_by_peer", ({ peerSocketId }) => {
            setRemotePeerDetails(prev => ({ ...prev, socketId: peerSocketId }));
            createPeerConnection(peerSocketId);
        });

        socket.on("receive_rtc_offer", async ({ fromSocketId, offer }) => {
            setCallState('connected');
            const pc = getOrCreatePeerConnection(fromSocketId);
            await pc.setRemoteDescription(new RTCSessionDescription(offer));
            const answer = await pc.createAnswer();
            await pc.setLocalDescription(answer);
            socket.emit("send_rtc_answer", { toSocketId: fromSocketId, answer });
        });

        socket.on("receive_rtc_answer", async ({ answer }) => {
            if (peerConnectionRef.current) {
                await peerConnectionRef.current.setRemoteDescription(new RTCSessionDescription(answer));
            }
        });

        socket.on("receive_ice_candidate", async ({ candidate }) => {
            if (peerConnectionRef.current && candidate) {
                try { await peerConnectionRef.current.addIceCandidate(new RTCIceCandidate(candidate)); }
                catch (e) { console.error(e); }
            }
        });

        socket.on("call_terminated_by_peer", () => {
            clearCallSessionState();
        });

        return () => {
            socket.off("receive_message"); socket.off("user_typing"); socket.off("message_read_update");
            socket.off("message_edited"); socket.off("message_deleted"); socket.off("room_users_update");
            socket.off("message_reaction_update"); socket.off("incoming_call_signal");
            socket.off("call_accepted_by_peer"); socket.off("receive_rtc_offer");
            socket.off("receive_rtc_answer"); socket.off("receive_ice_candidate"); socket.off("call_terminated_by_peer");
        };
    }, [socket, isOpen, currentUserId, currentUsername]);
    // ==========================================
    // LIVE CHAT DRAWER - PART 6 OF 9 (RTC METHODS)
    // ==========================================
    const clearCallSessionState = () => {
        setCallState('idle');
        if (localStreamRef.current) {
            localStreamRef.current.getTracks().forEach(track => track.stop());
            localStreamRef.current = null;
        }
        if (peerConnectionRef.current) {
            peerConnectionRef.current.close();
            peerConnectionRef.current = null;
        }
        setRemotePeerDetails({ socketId: '', name: '' });
        triggerToast("Call ended", "info");
    };

    const getOrCreatePeerConnection = (targetSocketId) => {
        if (peerConnectionRef.current) return peerConnectionRef.current;

        const pc = new RTCPeerConnection(RTC_CONFIG);

        pc.onicecandidate = (event) => {
            if (event.candidate && socket) {
                socket.emit("send_ice_candidate", { toSocketId: targetSocketId, candidate: event.candidate });
            }
        };

        pc.ontrack = (event) => {
            if (remoteVideoRef.current && event.streams[0]) {
                remoteVideoRef.current.srcObject = event.streams[0];
            }
        };

        if (localStreamRef.current) {
            localStreamRef.current.getTracks().forEach(track => pc.addTrack(track, localStreamRef.current));
        }

        peerConnectionRef.current = pc;
        return pc;
    };

    const createPeerConnection = async (targetSocketId) => {
        setCallState('connected');
        const pc = getOrCreatePeerConnection(targetSocketId);
        const offer = await pc.createOffer();
        await pc.setLocalDescription(offer);
        socket.emit("send_rtc_offer", { toSocketId: targetSocketId, offer });
    };
    // ==========================================
    // LIVE CHAT DRAWER - PART 7 OF 9 (CALL TRIGGERS)
    // ==========================================
    const initiateCallSession = async (type) => {
        setCallState('ringing');
        setCallType(type);
        try {
            const stream = await navigator.mediaDevices.getUserMedia({ audio: true, video: type === 'video' });
            localStreamRef.current = stream;
            if (localVideoRef.current) localVideoRef.current.srcObject = stream;

            socket.emit("initiate_call", { room: "global_workspace", callerName: currentUsername, type });
        } catch (err) {
            setCallState('idle');
            triggerToast("Camera/Microphone access denied", "error");
        }
    };

    const acceptIncomingCall = async () => {
        try {
            const stream = await navigator.mediaDevices.getUserMedia({ audio: true, video: callType === 'video' });
            localStreamRef.current = stream;
            if (localVideoRef.current) localVideoRef.current.srcObject = stream;

            socket.emit("accept_incoming_call", { toSocketId: remotePeerDetails.socketId });
        } catch (err) {
            socket.emit("end_call_session", { room: "global_workspace" });
            clearCallSessionState();
            triggerToast("Media permissions error", "error");
        }
    };

    const endCallSession = () => {
        socket.emit("end_call_session", { room: "global_workspace" });
        clearCallSessionState();
    };

    // Input listeners and Audio note captures
    const handleInputChange = (e) => {
        setMessage(e.target.value);
        if (!socket) return;
        socket.emit("typing", { room: "global_workspace", username: currentUsername, isTyping: true });
        clearTimeout(typingTimeoutRef.current);
        typingTimeoutRef.current = setTimeout(() => { socket.emit("typing", { room: "global_workspace", username: currentUsername, isTyping: false }); }, 2000);
    };

    const handleFileChange = (e) => {
        const file = e.target.files?.[0];
        if (!file) return;
        const reader = new FileReader();
        reader.onloadend = () => { setSelectedFile({ name: file.name, url: reader.result, type: file.type.startsWith('image/') ? 'image' : 'file' }); };
        reader.readAsDataURL(file);
    };

    const startRecording = async () => {
        try {
            const stream = await navigator.mediaDevices.getUserMedia({ audio: true });
            audioChunksRef.current = [];
            const mediaRecorder = new MediaRecorder(stream, { mimeType: 'audio/webm' });
            mediaRecorderRef.current = mediaRecorder;
            mediaRecorder.ondataavailable = (e) => { if (e.data.size > 0) audioChunksRef.current.push(e.data); };
            // 🛠️ Look for this block in your code and update the highlighted line:
            mediaRecorder.onstop = () => {
                const audioBlob = new Blob(audioChunksRef.current, { type: 'audio/webm' });
                const reader = new FileReader();
                reader.onloadend = () => {
                    setSelectedFile({ name: `Voice Note (${recordingDuration}s)`, url: reader.result, type: 'audio' });
                };
                reader.readAsDataURL(audioBlob);

                // CHANGED THIS LINE: Change track.stop() to t.stop()
                stream.getTracks().forEach(t => t.stop());
            };

            mediaRecorder.start();
            setIsRecording(true); setRecordingDuration(0);
            durationIntervalRef.current = setInterval(() => { setRecordingDuration(prev => prev + 1); }, 1000);
        } catch (err) { triggerToast("Microphone access denied", "error"); }
    };

    const stopRecording = () => { if (mediaRecorderRef.current && isRecording) { mediaRecorderRef.current.stop(); setIsRecording(false); clearInterval(durationIntervalRef.current); } };
    const handleSendMessage = (e) => { e.preventDefault(); if (!message.trim() && !selectedFile) return; const data = { room: 'global_workspace', senderId: currentUserId, senderName: currentUsername, message: message.trim(), type: selectedFile ? selectedFile.type : "text", attachment: selectedFile ? selectedFile.url : null, attachmentType: selectedFile ? selectedFile.name : null, createdAt: new Date() }; socket.emit('send_message', data); setMessage(''); setSelectedFile(null); socket.emit("typing", { room: "global_workspace", username: currentUsername, isTyping: false }); };
    const triggerSaveEdit = (messageId) => { if (!editInput.trim() || !socket) return; socket.emit("edit_message", { room: "global_workspace", messageId, newMessage: editInput.trim() }); setChatLog(prev => prev.map(m => m._id === messageId ? { ...m, message: editInput.trim() } : m)); setEditingMessageId(null); };
    const triggerDelete = (messageId) => { if (!socket) return; triggerToast(<div style={{ display: 'flex', alignItems: 'center', justifyContent: 'space-between', width: '100%', gap: '8px' }}><span>Delete message?</span><div style={{ display: 'flex', gap: '6px' }}><button onClick={() => { socket.emit("delete_message", { room: "global_workspace", messageId }); setChatLog(prev => prev.filter(m => m._id !== messageId)); triggerToast("Message deleted successfully", "success"); }} style={{ background: '#ef4444', color: '#fff', border: 'none', borderRadius: '4px', padding: '3px 8px', fontSize: '0.65rem', cursor: 'pointer', fontWeight: 'bold' }}>Delete</button><button onClick={() => setToast({ show: false, message: '', type: 'info' })} style={{ background: '#64748b', color: '#fff', border: 'none', borderRadius: '4px', padding: '3px 8px', fontSize: '0.65rem', cursor: 'pointer' }}>No</button></div></div>, "error"); };
    const triggerAddReaction = (messageId, emoji) => { if (!socket) return; socket.emit("send_reaction", { room: "global_workspace", messageId, emoji, username: currentUsername }); };
    // ==========================================
    // LIVE CHAT DRAWER - PART 8 OF 9 (LAYOUT)
    // ==========================================
    return (
        <div style={{ position: 'fixed', bottom: 0, right: '24px', zIndex: 9999, fontFamily: 'system-ui, sans-serif' }}>
            {!isOpen && (
                <button onClick={() => setIsOpen(true)} style={{ display: 'flex', alignItems: 'center', gap: '8px', padding: '12px 20px', background: '#2563eb', color: '#ffffff', border: 'none', borderRadius: '12px 12px 0 0', cursor: 'pointer', fontWeight: '700', boxShadow: '0 -4px 12px rgba(37,99,235,0.25)' }}>
                    Live Corporate Chat {activeUsersList.length > 0 && <span style={{ background: '#22c55e', color: '#fff', borderRadius: '10px', padding: '2px 6px', fontSize: '0.65rem', fontWeight: 'bold' }}>{activeUsersList.length}</span>}
                </button>
            )}
            {isOpen && (
                <div style={{ width: '360px', height: '520px', background: 'var(--bg-card, #ffffff)', border: '1px solid var(--border-color, #e2e8f0)', borderRadius: '12px 12px 0 0', boxShadow: '0 -12px 35px rgba(15,23,42,0.25)', display: 'flex', flexDirection: 'column', overflow: 'hidden', position: 'relative' }}>

                    {/* Header element overlay */}
                    <div style={{ padding: '14px 16px', background: 'linear-gradient(135deg, #1e40af, #2563eb)', color: '#ffffff', display: 'flex', justifyContent: 'space-between', alignItems: 'center', position: 'relative' }}>
                        <div>
                            <div style={{ display: 'flex', alignItems: 'center', gap: '8px', fontSize: '0.95rem', fontWeight: 800 }}><MessageCircle size={16} /> <span>Global Workspace</span></div>
                            <div style={{ fontSize: '0.7rem', opacity: 0.85 }}>Online ({activeUsersList.length}): {activeUsersList.slice(0, 2).join(', ')}</div>
                        </div>

                        {/* 📹 NEW: Audio / Video Calling Action Icons placed inside Header Toolbar */}
                        {callState === 'idle' && (
                            <div style={{ display: 'flex', gap: '8px', marginLeft: 'auto', marginRight: '12px' }}>
                                <button onClick={() => initiateCallSession('voice')} style={{ background: 'transparent', border: 'none', color: '#fff', cursor: 'pointer' }} title="Voice Call"><Phone size={16} /></button>
                                <button onClick={() => initiateCallSession('video')} style={{ background: 'transparent', border: 'none', color: '#fff', cursor: 'pointer' }} title="Video Call"><Video size={16} /></button>
                            </div>
                        )}
                        <button onClick={() => setIsOpen(false)} style={{ background: 'transparent', border: 'none', color: '#ffffff', cursor: 'pointer', fontSize: '14px' }}>✕</button>

                        {toast.show && (
                            <div style={{ position: 'absolute', top: '100%', left: 0, right: 0, padding: '10px 16px', color: '#ffffff', fontSize: '0.75rem', fontWeight: '600', zIndex: 100000, background: toast.type === 'success' ? '#22c55e' : toast.type === 'error' ? '#1e293b' : '#3b82f6', boxShadow: '0 8px 20px rgba(0,0,0,0.2)', borderBottom: toast.type === 'error' ? '2px solid #ef4444' : 'none', animation: 'chatFadeIn 0.2s ease', boxSizing: 'border-box' }}><div style={{ width: '100%' }}>{toast.message}</div></div>
                        )}
                    </div>

                    {/* 📹 NEW WEBRTC PEER CALL CONTROLLER POPUP UI BLOCKS */}
                    {callState !== 'idle' && (
                        <div style={{ position: 'absolute', top: '55px', left: 0, right: 0, bottom: 0, background: '#0f172a', zIndex: 99999, display: 'flex', flexDirection: 'column', alignItems: 'center', justifyContent: 'center', color: '#fff', padding: '16px' }}>
                            {callState === 'ringing' && <div style={{ textAlign: 'center' }}><h3>Calling Room Line...</h3><p style={{ fontSize: '0.8rem', opacity: 0.7 }}>Waiting for colleague peer lines...</p></div>}
                            {callState === 'incoming' && (
                                <div style={{ textAlign: 'center' }}>
                                    <h3>Incoming {callType} Call</h3>
                                    <p style={{ fontSize: '0.8rem', opacity: 0.7 }}>From: <b>{remotePeerDetails.name}</b></p>
                                    <div style={{ display: 'flex', gap: '12px', marginTop: '14px', justifyContent: 'center' }}>
                                        <button onClick={acceptIncomingCall} style={{ background: '#22c55e', color: '#fff', border: 'none', borderRadius: '20px', padding: '8px 16px', fontSize: '0.75rem', fontWeight: 'bold', cursor: 'pointer' }}>Answer</button>
                                        <button onClick={endCallSession} style={{ background: '#ef4444', color: '#fff', border: 'none', borderRadius: '20px', padding: '8px 16px', fontSize: '0.75rem', fontWeight: 'bold', cursor: 'pointer' }}>Decline</button>
                                    </div>
                                </div>
                            )}
                            {callState === 'connected' && (
                                <div style={{ width: '100%', height: '100%', position: 'relative', display: 'flex', flexDirection: 'column', gap: '8px' }}>
                                    {/* Live Streaming Video Frame Elements Viewport monitors */}
                                    {callType === 'video' ? (
                                        <div style={{ flex: 1, background: '#1e293b', borderRadius: '12px', overflow: 'hidden', position: 'relative' }}>
                                            <video ref={remoteVideoRef} autoPlay playsInline style={{ width: '100%', height: '100%', objectFit: 'cover' }} />
                                            <video ref={localVideoRef} autoPlay playsInline muted style={{ width: '90px', height: '120px', objectFit: 'cover', position: 'absolute', bottom: '8px', right: '8px', borderRadius: '8px', border: '2px solid #3b82f6', background: '#000' }} />
                                        </div>
                                    ) : (
                                        <div style={{ flex: 1, display: 'flex', alignItems: 'center', justifyContent: 'center', fontSize: '0.85rem' }}>🗣️ Connected Voice Call with {remotePeerDetails.name}</div>
                                    )}
                                    {/* Hidden fallback monitors for audio paths handling */}
                                    {callType === 'voice' && <video ref={localVideoRef} autoPlay playsInline muted style={{ display: 'none' }} />}
                                    <audio ref={remoteVideoRef} autoPlay style={{ display: 'none' }} />
                                </div>
                            )}
                            {/* Universal hang up active calls session button row controller */}
                            {callState !== 'incoming' && (
                                <button onClick={endCallSession} style={{ background: '#ef4444', color: '#fff', border: 'none', width: '42px', height: '42px', borderRadius: '50%', display: 'flex', alignItems: 'center', justifyContent: 'center', marginTop: '12px', cursor: 'pointer', boxShadow: '0 4px 10px rgba(239,68,68,0.3)' }} title="Hang Up"><PhoneOff size={18} /></button>
                            )}
                        </div>
                    )}

                    {/* Messages Feed View Stream Logs Container mapping */}
                    <div style={{ flex: 1, padding: '14px', overflowY: 'auto', display: 'flex', flexDirection: 'column', gap: '10px', background: 'var(--bg-main, #f8fafc)' }}>
                        {chatLog.map((msg, idx) => {
                            const isMe = msg.senderName === user?.username || msg.senderName === user?.name;
                            const isSame = chatLog[idx - 1] && chatLog[idx - 1].senderName === msg.senderName;
                            const isHovered = hoveredMessageId === msg._id;
                            const activeReactions = reactionsMap[msg._id] || [];
                            return (
                                <div key={msg._id || idx} onMouseEnter={() => setHoveredMessageId(msg._id)} onMouseLeave={() => setHoveredMessageId(null)} style={{ display: 'flex', flexDirection: isMe ? 'row' : 'row-reverse', alignItems: 'flex-end', gap: '8px', alignSelf: isMe ? 'flex-start' : 'flex-end', maxWidth: '85%', marginTop: isSame ? '-4px' : '8px', position: 'relative' }}>
                                    <div style={{ width: 30, height: 30, flexShrink: 0, visibility: isSame ? 'hidden' : 'visible', borderRadius: '50%', background: isMe ? '#2563eb' : '#64748b', color: '#fff', display: 'flex', alignItems: 'center', justifyContent: 'center', fontWeight: 700, fontSize: '0.7rem' }}>{(msg.senderName || '?').charAt(0).toUpperCase()}</div>
                                    <div style={{ display: 'flex', flexDirection: 'column', alignItems: isMe ? 'flex-start' : 'flex-end' }}>
                                        {!isSame && <div style={{ display: 'flex', gap: '6px', marginBottom: '4px', fontSize: '0.7rem' }}><span style={{ fontWeight: 700, color: 'var(--text-primary, #334155)' }}>{msg.senderName}</span></div>}
                                        <div style={{ padding: '8px 12px', borderRadius: isSame ? '14px' : isMe ? '18px 18px 18px 4px' : '18px 18px 4px 18px', background: isMe ? 'linear-gradient(135deg, #2563eb, #3b82f6)' : 'var(--bg-card, #ffffff)', color: isMe ? '#ffffff' : 'var(--text-primary, #0f172a)', fontSize: '0.75rem', border: isMe ? 'none' : '1px solid var(--border-color, #cbd5e1)', boxShadow: '0 2px 8px rgba(0,0,0,0.04)', position: 'relative' }}>
                                            {msg.type === 'image' && msg.attachment && <img src={msg.attachment} alt="Attachment" style={{ maxWidth: '100%', maxHeight: '140px', borderRadius: '8px', marginBottom: '6px', display: 'block' }} />}
                                            {msg.type === 'file' && msg.attachment && <a href={msg.attachment} download={msg.attachmentType || "download"} style={{ display: 'flex', alignItems: 'center', gap: '6px', color: isMe ? '#fff' : '#2563eb', fontWeight: 'bold', textDecoration: 'underline', marginBottom: '6px' }}>📁 {msg.attachmentType || 'Download File'}</a>}
                                            {msg.type === 'audio' && msg.attachment && <audio src={msg.attachment} controls style={{ maxWidth: '200px', height: '30px', marginTop: '4px', display: 'block' }} />}
                                            {editingMessageId === msg._id ? (
                                                <div style={{ display: 'flex', flexDirection: 'column', gap: '6px', minWidth: '180px', marginTop: '4px', background: 'rgba(0,0,0,0.03)', padding: '6px', borderRadius: '8px' }}><input type="text" value={editInput} onChange={(e) => setEditInput(e.target.value)} style={{ width: '100%', padding: '6px 8px', fontSize: '0.75rem', borderRadius: '6px', border: '1px solid #cbd5e1', background: '#fff', color: '#000', boxSizing: 'border-box' }} /><div style={{ display: 'flex', justifyContent: 'flex-end', gap: '4px' }}><button onClick={() => setEditingMessageId(null)} style={{ display: 'flex', alignItems: 'center', gap: '3px', padding: '4px 8px', fontSize: '0.65rem', background: '#64748b', color: '#fff', border: 'none', borderRadius: '4px', cursor: 'pointer' }}><X size={10} /> Cancel</button><button onClick={() => triggerSaveEdit(msg._id)} style={{ display: 'flex', alignItems: 'center', gap: '3px', padding: '4px 8px', fontSize: '0.65rem', background: '#22c55e', color: '#fff', border: 'none', borderRadius: '4px', cursor: 'pointer' }}><Check size={10} /> Save</button></div></div>
                                            ) : (
                                                msg.type === 'text' && <div style={{ display: 'flex', gap: '8px', alignItems: 'flex-end', justifyContent: 'space-between' }}><span>{msg.message}</span><span style={{ fontSize: '0.6rem', opacity: 0.65, whiteSpace: 'nowrap', display: 'flex', alignItems: 'center' }}>{new Date(msg.createdAt).toLocaleTimeString([], { hour: "2-digit", minute: "2-digit" })}{isMe && <span dangerouslySetInnerHTML={{ __html: (msg.readBy?.length > 1 ? '<span style="color:#25d366;font-size:0.85rem;margin-left:5px;font-weight:900">✓✓</span>' : '<span style="color:var(--text-muted,#94a3b8);font-size:0.85rem;margin-left:5px">✓</span>') }} />}</span></div>
                                            )}
                                            {activeReactions.length > 0 && (
                                                <div style={{ display: 'flex', gap: '2px', position: 'absolute', bottom: '-12px', [isMe ? 'left' : 'right']: '8px', background: 'var(--bg-card, #fff)', border: '1px solid var(--border-color, #cbd5e1)', borderRadius: '8px', padding: '1px 4px', fontSize: '0.65rem', boxShadow: '0 2px 4px rgba(0,0,0,0.06)', zIndex: 5 }}>{Object.entries(activeReactions.reduce((acc, curr) => ({ ...acc, [curr.emoji]: (acc[curr.emoji] || 0) + 1 }), {})).map(([emo, count]) => (<span key={emo}>{emo}{count > 1 ? count : ''}</span>))}</div>
                                            )}
                                            {isHovered && !editingMessageId && (<div style={{ position: 'absolute', top: '-28px', [isMe ? 'right' : 'left']: '4px', display: 'flex', alignItems: 'center', gap: '6px', background: 'var(--bg-card, #ffffff)', padding: '3px 6px', borderRadius: '8px', border: '1px solid var(--border-color, #cbd5e1)', boxShadow: '0 4px 10px rgba(0,0,0,0.12)', zIndex: 10 }}><div style={{ display: 'flex', gap: '3px', borderRight: isMe ? '1px solid var(--border-color, #e2e8f0)' : 'none', paddingRight: isMe ? '4px' : 0 }}>{REACTION_OPTIONS.map(emo => (<button key={emo} type="button" onClick={() => triggerAddReaction(msg._id, emo)} style={{ background: 'transparent', border: 'none', padding: '1px', fontSize: '0.85rem', cursor: 'pointer' }}>{emo}</button>))}</div>{isMe && (<div style={{ display: 'flex', gap: '4px' }}><button type="button" onClick={() => { setEditingMessageId(msg._id); setEditInput(msg.message); }} style={{ background: 'transparent', border: 'none', color: '#2563eb', cursor: 'pointer', padding: '2px' }}><Edit2 size={11} /></button><button type="button" onClick={() => triggerDelete(msg._id)} style={{ background: 'transparent', border: 'none', color: '#ef4444', cursor: 'pointer', padding: '2px' }}><Trash2 size={11} /></button></div>)}</div>)}
                                        </div>
                                    </div>
                                </div>
                            );
                        })}
                        {typingUser && <div style={{ fontSize: '0.7rem', color: 'var(--text-muted, #64748b)', fontStyle: 'italic', paddingLeft: '38px' }}>✍️ {typingUser} is typing...</div>}
                        <div ref={messageEndRef} />
                    </div>
                    {showEmojiPicker && <div style={{ position: 'absolute', bottom: '65px', right: '10px', zIndex: 10000 }}><EmojiPicker onEmojiClick={(ed) => { setMessage(p => p + ed.emoji); setShowEmojiPicker(false); }} width={300} height={320} searchDisabled previewConfig={{ showPreview: false }} /></div>}
                    {selectedFile && <div style={{ background: 'var(--bg-main, #eff6ff)', padding: '6px 12px', fontSize: '0.7rem', borderTop: '1px solid var(--border-color, #dbeafe)', display: 'flex', justifyContent: 'space-between', color: '#1e40af' }}><span>📎 Staged: <b>{selectedFile.name}</b></span><button onClick={() => setSelectedFile(null)} style={{ border: 'none', background: 'transparent', color: '#ef4444', cursor: 'pointer' }}>Remove</button></div>}

                    <form onSubmit={handleSendMessage} style={{ padding: '12px', display: 'flex', gap: '6px', alignItems: 'center', justifyContent: 'center', borderTop: '1px solid var(--border-color, #e2e8f0)', background: 'var(--bg-card, #ffffff)', boxSizing: 'border-box' }} >
                        <div style={{ width: '28px', height: '38px', display: 'flex', alignItems: 'center', justifyContent: 'center' }}><button type="button" onClick={() => setShowEmojiPicker(!showEmojiPicker)} style={{ border: 'none', background: 'transparent', color: 'var(--text-muted, #64748b)', cursor: 'pointer', padding: 0, display: 'flex', alignItems: 'center', justifyContent: 'center' }}><Smile size={20} /></button></div>
                        <div style={{ width: '28px', height: '38px', display: 'flex', alignItems: 'center', justifyContent: 'center' }}><label style={{ cursor: 'pointer', color: 'var(--text-muted, #64748b)', padding: 0, display: 'flex', alignItems: 'center', justifyContent: 'center' }}><Paperclip size={20} /><input type="file" onChange={handleFileChange} style={{ display: 'none' }} /></label></div>
                        <div style={{ width: '28px', height: '38px', display: 'flex', alignItems: 'center', justifyContent: 'center' }}>
                            {isRecording ? (
                                <button type="button" onClick={stopRecording} style={{ border: 'none', background: 'transparent', color: '#ef4444', cursor: 'pointer', padding: 0, display: 'flex', alignItems: 'center' }} title={`Stop (${recordingDuration}s)`}><Square size={18} /></button>
                            ) : (
                                <button type="button" onClick={startRecording} style={{ border: 'none', background: 'transparent', color: 'var(--text-muted, #64748b)', cursor: 'pointer', padding: 0, display: 'flex', alignItems: 'center' }} title="Record Voice"><Mic size={18} /></button>
                            )}
                        </div>
                        <input type="text" value={message} onChange={handleInputChange} placeholder={isRecording ? `Recording... ${recordingDuration}s` : "Write a message..."} disabled={isRecording} style={{ flex: 1, height: '38px', padding: '0 12px', borderRadius: '12px', border: '1px solid var(--border-color, #cbd5e1)', background: 'var(--bg-main, #f8fafc)', color: 'var(--text-primary, #0f172a)', outline: 'none', fontSize: '0.8rem', boxSizing: 'border-box' }} />
                        <button
                            type="submit"
                            disabled={isRecording}
                            style={{ width: '38px', height: '38px', border: 'none', borderRadius: '12px', background: 'linear-gradient(135deg, #2563eb, #3b82f6)', color: '#ffffff', cursor: 'pointer', display: 'flex', alignItems: 'center', justifyContent: 'center', flexShrink: 0, boxSizing: 'border-box' }}
                        >
                            <Send size={15} />
                        </button>
                    </form>
                </div>
            )}
        </div>
    );
};
